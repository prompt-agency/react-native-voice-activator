```typescript
export interface ModelAssetsProps {
  speakerModelPath?: string;
  denoiserModelPath?: string;
}

const MODEL_FIELDS = [
  { field: 'speakerModelPath' as const, subdir: 'SherpaOnnxSpeaker' },
  { field: 'denoiserModelPath' as const, subdir: 'SherpaOnnxDenoiser' },
] as const;

function resolveModelPath(
  projectRoot: string,
  rawPath: string,
  fieldName: string
): string {
  const resolved = path.isAbsolute(rawPath)
    ? rawPath
    : path.resolve(projectRoot, rawPath);
  if (!existsSync(resolved)) {
    throw new Error(
      `[react-native-voice-activator] ${fieldName}: configured path does not exist: ${resolved}`
    );
  }
  return resolved;
}

export const withModelAssets: ConfigPlugin<ModelAssetsProps> = (config, props) => {
  // iOS withDangerousMod: copies to platformProjectRoot/appName/subdir/filename
  // iOS withXcodeProject: registers in .pbxproj via addResourceFile
  // Android withDangerousMod: copies to platformProjectRoot/app/src/main/assets/voice-activator-sherpa-onnx/filename
};
```

**Approach:** Since `withModelAssets` is a ConfigPlugin that calls `withDangerousMod` and `withXcodeProject`, the cleanest test strategy is to:
1. Mock `node:fs` (`existsSync`, `mkdirSync`, `copyFileSync`)
2. Mock `@expo/config-plugins` (`withDangerousMod`, `withXcodeProject`) to capture and invoke the callbacks synchronously
3. Assert that `copyFileSync` was called with the correct source and destination paths
4. Assert that `mkdirSync` was called with `{ recursive: true }`
5. Assert that `addResourceFile` was called with the correct relative path

**Test file structure:**

```typescript
import { copyFileSync, existsSync, mkdirSync } from 'node:fs';
import path from 'node:path';

// Mock node:fs
jest.mock('node:fs', () => ({
  existsSync: jest.fn(),
  mkdirSync: jest.fn(),
  copyFileSync: jest.fn(),
}));

// Mock @expo/config-plugins to capture and invoke callbacks
const mockAddResourceFile = jest.fn();
const mockGetFirstTarget = jest.fn(() => ({ uuid: 'target-uuid' }));

jest.mock('@expo/config-plugins', () => ({
  withDangerousMod: jest.fn(
    (config: unknown, [_platform, callback]: [string, (mod: unknown) => unknown]) => {
      // Invoke the callback with a fake modConfig so assertions can check fs calls
      const modConfig = {
        modRequest: {
          projectRoot: '/project',
          platformProjectRoot: _platform === 'ios' ? '/project/ios' : '/project/android',
          projectName: 'MyApp',
        },
      };
      callback(modConfig);
      return config;
    }
  ),
  withXcodeProject: jest.fn(
    (config: unknown, callback: (mod: unknown) => unknown) => {
      const modConfig = {
        modRequest: { projectName: 'MyApp' },
        modResults: {
          addResourceFile: mockAddResourceFile,
          getFirstTarget: mockGetFirstTarget,
        },
      };
      callback(modConfig);
      return config;
    }
  ),
}));

import { withModelAssets } from '../expo/withModelAssets';
```
